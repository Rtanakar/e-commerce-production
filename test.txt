// ============================================================================
// schema.prisma - Database schema (Postgres + Prisma)
// ============================================================================
// Multi-vendor e-commerce ka data model
// Iteratively grow karenge - abhi auth + vendor + address basics
// ============================================================================

generator client {
  provider        = "prisma-client-js"
  previewFeatures = ["driverAdapters", "fullTextSearchPostgres"]
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

// ============================================================================
// ENUMS
// ============================================================================

enum UserRole {
  CUSTOMER  // normal buyer
  VENDOR    // seller (multi-vendor like Amazon sellers)
  ADMIN     // platform admin
}

enum UserStatus {
  ACTIVE
  SUSPENDED
  DELETED
}

// ============================================================================
// USER
// ============================================================================
// Core authentication entity
// ============================================================================
model User {
  // cuid - collision-resistant, sorted, shorter than UUID
  id            String     @id @default(cuid())
  email         String     @unique
  // bcrypt hash - 60 chars, salt embedded
  password      String
  name          String?
  phone         String?    @unique
  emailVerified DateTime?
  phoneVerified DateTime?
  // Cloudinary/ImageKit URL
  image         String?
  role          UserRole   @default(CUSTOMER)
  status        UserStatus @default(ACTIVE)

  // Soft delete - compliance/audit ke liye preserve
  deletedAt DateTime?

  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  // Relations
  vendorProfile VendorProfile?
  addresses     Address[]

  // Indexes - common queries pe performance
  @@index([email])
  @@index([status, role])
  @@map("users")
}

// ============================================================================
// VENDOR PROFILE
// ============================================================================
// 1-to-1 with User - alag table for separation of concerns
// User table me sab kuch daalna bad design hai
// ============================================================================
model VendorProfile {
  id          String  @id @default(cuid())
  userId      String  @unique
  user        User    @relation(fields: [userId], references: [id], onDelete: Cascade)
  shopName    String  @unique
  shopSlug    String  @unique
  description String?
  // GST mandatory India me vendor ke liye
  gstNumber   String?
  // Admin verify karega tabhi sell kar sakta
  isVerified  Boolean @default(false)
  verifiedAt  DateTime?

  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  @@index([shopSlug])
  @@index([isVerified])
  @@map("vendor_profiles")
}

// ============================================================================
// ADDRESS
// ============================================================================
// Multiple addresses per user - shipping
// ============================================================================
model Address {
  id        String  @id @default(cuid())
  userId    String
  user      User    @relation(fields: [userId], references: [id], onDelete: Cascade)
  fullName  String
  phone     String
  line1     String
  line2     String?
  city      String
  state     String
  pincode   String
  country   String  @default("IN")
  isDefault Boolean @default(false)

  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  @@index([userId])
  @@map("addresses")
}







// ============================================================================
// seed.ts - Database seeding
// ============================================================================
// Local dev me test data daalne ke liye
// `pnpm db:seed` / `npm run db:seed`
// ============================================================================

import { PrismaClient, UserRole } from "@prisma/client";
import bcrypt from "bcryptjs";

const prisma = new PrismaClient();

async function main() {
  console.log("🌱 Seeding database...");

  // Admin user - local dev only
  // Production me NEVER hardcoded creds
  const adminPassword = await bcrypt.hash("Admin@12345", 12);
  await prisma.user.upsert({
    where: { email: "admin@shop.local" },
    update: {},
    create: {
      email: "admin@shop.local",
      password: adminPassword,
      name: "Platform Admin",
      role: UserRole.ADMIN,
      emailVerified: new Date(),
    },
  });

  console.log("✅ Seed complete");
  console.log("   Admin: admin@shop.local / Admin@12345");
}

main()
  .catch((e) => {
    console.error(e);
    process.exit(1);
  })
  .finally(async () => {
    await prisma.$disconnect();
  });
